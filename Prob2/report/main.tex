\section*{Exercise 1}
Quantum phase estimation (QPE) can be used to measure the Hamming weight of a quantum state and prepare Dicke states.

\subsubsection*{a) Write down the Hamiltonian whose eigenvalues can distinguish the Hamming weight of an n qubit quantum state.}

The Hamiltonian whose eigenvalues differs by Hamming weight in $n$-qubit system can be constructed by the projector onto the space of fixed Hamming weight $i$. That is,
$$
\mathcal{H} = \sum_{i=0}^{2^n -1} e_i P_{i}
$$
We can choose $e_i = i$ where the eigenvalue becomes the Hamming weight $i$.
Note that the given hamiltonian can have ${}_{n}C_{i}$ for $e_i$.

\subsubsection*{b) For an $n$ qubit state, what is the minimum number of bits needed to measure the hamming weight of the state? Hint: The Hamming weight of the state is in the range $\{0,1,..,n\}$}
Let $k$ denote the minimum number of bits required to measure hamming weight of the state. Then
$$
k = \left[\log_{2}(n+1)\right]
$$
where $\left[ a \right]$ denotes the smallest integer larger than $a$.

\section*{Exercise 2}
Using qiskit, create a function that takes in a quantum state in vector format as input and returns a quantum circuit that measures the Hamming weight of the state. Hint: see ref \cite{rethinasamy2024logarithmicdepthquantumcircuitshamming}. You can use qiskit's initialize module to initialize the states.

From the given reference, in order to measure Hamming weight of $n$ qubit system, it requires $k$ qubits as a control register as discussed in the Exercise 1 b). 
Let $\ket{\Psi}$ denote the wavefunction for whole system including control registers and prepared state.
Given arbitrary state $\ket{\psi}$, the following algorithm starts with preparing $\ket{\psi} \otimes \ket{0}_{C}^{\otimes k}$.
Then, act on the control register qubits with $k$ Hadamard gates, we have:
\begin{align}
	\ket{\Psi} &= \ket{\psi} \otimes H^{\otimes k} \ket{0}_{C}^{\otimes k} \\
	&= \ket{\psi} \otimes \ket{+}_{C}^{\otimes k}
\end{align} 
Next, acting on $R_{z}(\gamma_{j})_{C_{j}}$ gate on $j$ qubit, labeled $C_{j}$ where $\gamma_{j} = \frac{n\pi}{N+1}2^{j-1}$.
\begin{align}
	\ket{\Psi} = \ket{\psi} \otimes \Pi_{j=1}^{k} R_{z}(\gamma_{j})_{C_{j}} \ket{+}_{C_{j}}
	\label{eq:algorithm2}
\end{align}
Applying the following controlled-Z gate,
\begin{align}
	U = \Pi_{j=1}^{k} \left[ \ketbra{0}_{C_j} \otimes I_S + \ketbra{1}_{C_j} \otimes R_z^{\otimes n} (\beta_j)_S\right]
	\label{eq:algorithm3}
\end{align}
where $\beta_j = \frac{2\pi}{N+1} 2^{j-1}$.
Finally, IQFT(inverse quantum Fourier Transform) on register qubits and measure register qubits in the computational basis $\left\{ \ketbra{0}, \ketbra{1}, \cdots, \ketbra{N} \right\}$.

Let me show a detailed process that how the state changes that enables measurement of Hamming weight.
After the Eq. \ref{eq:algorithm2}, the state becomes:
\begin{align}
    \ket{y}_C &\rightarrow \bigotimes_{j=1}^k R_z(\gamma_j)_{C_j} \ket{y_k}_{C_k} \ldots \ket{y_1}_{C_1} \notag \\
    &= \exp{\!\left( \sum_{j=1}^k \frac{\mathrm{i} \gamma_j}{2}(2y_j - 1) \right)} \ket{y_k}_{C_k} \ldots \ket{y_1}_{C_1} \notag \\
    &= \exp\!\left( \sum_{j=1}^k \mathrm{i} \gamma_j y_j - \frac{\mathrm{i}}{2} \sum_{j=1}^k\gamma_j \right)\ket{y_k}_{C_k} \ldots \ket{y_1}_{C_1} \notag \\
    &= \exp\!\left( \mathrm{i} \vec{\gamma}\cdot\vec{y}  - \frac{\mathrm{i}}{2} \Gamma \right) \ket{y}_C,
\end{align}
where $\Gamma = \sum_{j=1}^k \gamma_j$.
Thus, 
\begin{align}
	\ket{\Psi} = \ket{\psi} \otimes \frac{1}{\sqrt{N+1}} \sum_{y=0}^N \exp{\!\left( \mathrm{i} \vec{\gamma}\cdot\vec{y}  - \frac{\mathrm{i}}{2} \Gamma \right)} \ket{y}_C
\end{align} 
Next, considering the action of controlled-Z gate in Eq. \ref{eq:algorithm3},
\begin{align}
	\ket{y}_C \ket{\psi}_S 
    &\rightarrow \ket{y}_C R_z^{\otimes n}(y_1 \beta_1) \cdots R_z^{\otimes n}(y_k \beta_k) \ket{\psi}_S \notag \\
    &= \ket{y}_C R_z^{\otimes n} (y_1\beta_1 + \cdots + y_k \beta_k) \ket{\psi}_S \notag \\
    &= \ket{y}_C R_z^{\otimes n} \!\left(\sum_{j=1}^k y_j\beta_j \right) \ket{\psi}_S \notag \\
    &= \ket{y}_C R_z^{\otimes n} \!\left(\vec{y}\cdot\vec{\beta} \right) \ket{\psi}_S,
\end{align}
thus, we have:
\begin{align}
	\ket{\Psi} = R_z^{\otimes n} \left(\vec{y}\cdot \vec{\beta}\right) \ket{\psi}_S \otimes \frac{1}{\sqrt{N+1}} \sum_{y=0}^N \exp{\!\left( \mathrm{i} \vec{\gamma}\cdot\vec{y}  - \frac{\mathrm{i}}{2} \Gamma \right)} \ket{y}_C
\end{align}
Here, 
\begin{align}
    \vec{\gamma}\cdot\vec{y} &= \sum_{j=1}^k \gamma_j y_j =\frac{n \pi}{N+1} \sum_{j=1}^k y_j 2^{j-1} = \frac{n \pi y}{N+1}, \\
    \vec{\beta}\cdot\vec{y} &= \sum_{j=1}^k \beta_j y_j =\frac{2 \pi}{N+1} \sum_{j=1}^k y_j 2^{j-1} = \frac{2 \pi y}{N+1}.
\end{align}
\begin{equation}
	\ket{\Psi} = \ket{y}_C \\ \otimes R_z^{\otimes n} \!\left( \frac{2 \pi y}{N+1}\right) \ket{\psi}_S \otimes \frac{1}{\sqrt{N+1}} \exp{\!\left(  - \frac{\mathrm{i}}{2} \Gamma \right)} \sum_{y=0}^N \exp{\!\left( \frac{\mathrm{i}n\pi y}{N+1} \right)} .
\end{equation}
Noting that,
\begin{equation}
    R_z^{\otimes n}\!\left(\frac{2 \pi y}{N+1}\right) = \exp\! \left(\frac{-\mathrm{i}n \pi y}{N+1} \right) \sum_{x=0}^n \exp\! \left( \frac{2\pi \mathrm{i} xy}{N+1} \right) P_x.
\end{equation}
after the controlled-Z operation, we see the $\ket{\Psi}$ can be expressed as:
\begin{equation}
	\ket{\Psi} = \frac{1}{\sqrt{N+1}} \exp{\!\left(  - \frac{\mathrm{i}}{2} \Gamma \right)} \sum_{y=0}^N \sum_{x=0}^n \exp\! \left( \frac{2\pi \mathrm{i} xy}{N+1} \right) P_x\ket{\psi}_S\ket{y}_C .
\end{equation} 
After applying IQFT on control register qubits, we have
\begin{equation}
	\frac{1}{N+1} \exp{\!\left(  - \frac{\mathrm{i}}{2} \Gamma \right)} \sum_{z,y,x=0}^N  \exp\! \left( \frac{2\pi \mathrm{i} (x-z) y}{N+1} \right) \\ \ket{z}_C P_x \ket{\psi}_S.
\end{equation}
Here the summation, 
\begin{align}
	S = \sum_{y=0}^N  \exp\! \left( \frac{2\pi \mathrm{i} (x-z) y}{N+1} \right) = (N+1)\delta_{xz}
\end{align}
The state finally becomes:
\begin{equation}
	\exp{\!\left(  - \frac{\mathrm{i}}{2} \Gamma \right)} \sum_{x=0}^N \ket{x}_C \otimes P_x\ket{\psi}_S
\end{equation}
Therefore, probability that we measure Hamming weight $a$ is:
\begin{equation}
	p_{a} = \left| P_{a} \ket{\psi} \right|^{2}
	\label{eq:hamming_weight_probability}
\end{equation}
and the post-measurement state $\ket{\bar{\psi}}$ is:
\begin{equation}
	\ket{\bar{\psi}}  = \frac{P_{a} \ket{\psi}}{\sqrt{p_{a}}}.
	\label{eq:post_measurement_state}
\end{equation}

\section*{Exercise 4}
\subsubsection*{Explain how the function in part 2 above can be used to prepare a Dicke state.}

The post-measurement state can be used to prepare Dicke state. From the Eq. \ref{eq:post_measurement_state}, 
we see that the initially given state projected on to the space of fixed Hamming weight $x$, therefore, becomes a linear combination computational bases of Hamming weight $x$.
For example of three-qubit system, and $x=2$, the basis is $\{\ket{011}, \ket{101}, \ket{110}\}$. 
Here Dicke state in $n$-qubit system and Hamming weight $k$ has form:
\begin{equation}
	\ket{D_{n,k}} = \frac{1}{\sqrt{{}_n C_{k}}} \sum_{i=1}^{{}_n C_{k}} \ket{e_{i}}
\end{equation}
where $\ket{e_i}$ is an element of basis of Hamming weight $k$ whose the number of basis is ${}_n C_{k}$.
Considering the whole coefficients are equal, the initial state before the measuring Hamming weight also has the same coefficients in the basis of computational basis.
Such state is just $+$ state:
\begin{equation}
	\ket{\psi} = \ket{++ \cdots +} = \frac{1}{\sqrt{2^n}} \sum_{i=0}^{2^n -1} \ket{i}
\end{equation}
where $i$ denotes the bit string. After measuring Hamming weight $k$, the post-measurement state $\ket{\bar{\psi}}$ becomes Dicke state $\ket{D_{n,k}}$.

\section*{Exercise 5}
Suppose we want to prepare the Dicke state with $n = 100$ and $k = 50$.

\subsubsection*{What is the probability that the state will be measured?}
Preparing + state, $\ket{\psi} = \ket{+}^{\otimes n}$, the probability that we measure Hamming weight $k$ is given in the Eq. \ref{eq:hamming_weight_probability}.
Among the computational basis, $\{\ket{0}, \ket{1}, \cdots, \ket{2^n}-1\}$, the number of basis whose Hamming weight is ${}_n C_{k}$. Then,
\begin{equation}
	p_{k} = \left| P_{k} \ket{\psi} \right|^{2} = \left| \frac{{}_n C_{k}}{\sqrt{2^n}} \right|^{2} = \frac{({}_n C_k)^{2}}{2^n}.
\end{equation}

\subsubsection*{How many gates are required?}

In phase 1, preparing control register C, consiting of $k$ qubits in initial state $\ket{0}_C$. Here we also should prepare $\ket{\psi}$ in system qubits. 
As discussed in Exercise 4, we prepare $\ket{+}$ state in system qubits, then we need $n$ Hadamard gate in this phase. 
In phase 2, acting on $k$ hadamard gates on control registers, we need $k$ Hadamard gate in this phase.
In phase 3, Acting on Rz gate on control register qubits, requires $k$ Rz gate. 
In phase 4, We perform controlled unitary gate in Eq. \ref{eq:algorithm3}, requires $n$ Controlled RZ gate per register qubit, therefore, we need $kn$ controlled Rz gates.
In phase 5, performing IQFT on controlled register qubits. It requires $k$ hadamard gate and it applies controlled Rz gate, control qubit in $j$ to target qubit $i$, where $i<j \leq k$ and $i$ runs from 0 to $k$. Therefore,  $k$ hadamard gate and $\frac{k(k-1)}{2}$ controlled Rz gate. 
Therefore, we need $2k + N$ Hadamard gates, $\frac{k(k-1)}{2} + nk$ controlled Z gate, and $k$ Rz gates.
