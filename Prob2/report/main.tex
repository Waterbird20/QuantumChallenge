\section*{Exercise 1}
Quantum phase estimation (QPE) can be used to measure the Hamming weight of a quantum state and prepare Dicke states.

\subsubsection*{a) Write down the Hamiltonian whose eigenvalues can distinguish the Hamming weight of an n qubit quantum state.}

The Hamiltonian whose eigenvalues distinguish the Hamming weight in an $n$-qubit system can be constructed as a sum of projectors onto the subspaces of fixed Hamming weight $i$, $P_{i}$ introduced in \cite{rethinasamy2024logarithmicdepthquantumcircuitshamming}. That is,
\begin{equation}
	\mathcal{H} = \sum_{i=0}^{n} e_i P_{i}
\end{equation}
where we can set $e_i = i$, so that the eigenvalue corresponds to the Hamming weight $i$. 
Note that the Hamiltonian can exhibit a degeneracy of ${}_{n}C_i$ for each eigenvalue $e_i$.

\subsubsection*{b) For an $n$ qubit state, what is the minimum number of bits needed to measure the hamming weight of the state? Hint: The Hamming weight of the state is in the range $\{0,1,..,n\}$}
Let $k$ represent the minimum number of bits required to measure the Hamming weight of an $n$-qubit state. Then,
\begin{equation}
	k = \lceil\log_{2}(n+1)\rceil
\end{equation}
where $\lceil a \rceil$ denotes the ceiling function, or the smallest integer greater than or equal to $a$.

Below, I provide the algorithm described in the reference. Please see the detailed calculation in the reference\cite{rethinasamy2024logarithmicdepthquantumcircuitshamming}.
% Todo Fix this
\paragraph{Phase 1. }Let $\ket{\Psi}$ represent the total system, including both the control registers and the prepared state.
Given an arbitrary state $\ket{\psi}$, the algorithm begins by preparing $\ket{\psi} \otimes \ket{0}_{C}^{\otimes k}$.
Next, apply $k$ Hadamard gates to the control register qubits:
\begin{align}
	\ket{\Psi} &= \ket{\psi} \otimes H^{\otimes k} \ket{0}_{C}^{\otimes k} \\
	&= \ket{\psi} \otimes \ket{+}_{C}^{\otimes k}
\end{align} 
\paragraph{Phase 2. }Apply $R_{z}(\gamma_{j}){C{j}}$ gates on each qubit $C_{j}$, where $\gamma_{j} = \frac{n\pi}{N+1}2^{j-1}$:
\begin{align}
	\ket{\Psi} = \ket{\psi} \otimes \Pi_{j=1}^{k} R_{z}(\gamma_{j})_{C_{j}} \ket{+}_{C_{j}}
	\label{eq:algorithm2}
\end{align}
\paragraph{Phase 3. }Apply the following controlled-Z gates:
\begin{align}
	U = \Pi_{j=1}^{k} \left[ \ketbra{0}_{C_j} \otimes I_S + \ketbra{1}_{C_j} \otimes R_z^{\otimes n} (\beta_j)_S\right]
	\label{eq:algorithm3}
\end{align}
where $\beta_j = \frac{2\pi}{N+1} 2^{j-1}$.
\paragraph{Phase 4. }Perform the inverse quantum Fourier transform (IQFT) on the control qubits and measure them in the computational basis $\left\{ \ketbra{0}, \ketbra{1}, \cdots, \ketbra{N} \right\}$.

\paragraph{State Evolutions}
Let me show a detailed process that how the state changes that enables measurement of Hamming weight.
After the Eq. \ref{eq:algorithm2}, the state evolves as:
\begin{align}
    \ket{y}_C &\rightarrow \bigotimes_{j=1}^k R_z(\gamma_j)_{C_j} \ket{y_k}_{C_k} \ldots \ket{y_1}_{C_1} \notag \\
    &= \exp{\!\left( \sum_{j=1}^k \frac{\mathrm{i} \gamma_j}{2}(2y_j - 1) \right)} \ket{y_k}_{C_k} \ldots \ket{y_1}_{C_1} \notag \\
    &= \exp\!\left( \sum_{j=1}^k \mathrm{i} \gamma_j y_j - \frac{\mathrm{i}}{2} \sum_{j=1}^k\gamma_j \right)\ket{y_k}_{C_k} \ldots \ket{y_1}_{C_1} \notag \\
    &= \exp\!\left( \mathrm{i} \vec{\gamma}\cdot\vec{y}  - \frac{\mathrm{i}}{2} \Gamma \right) \ket{y}_C,
\end{align}
where $\Gamma = \sum_{j=1}^k \gamma_j$.
Thus, the state becomes:
\begin{align}
	\ket{\Psi} = \ket{\psi} \otimes \frac{1}{\sqrt{N+1}} \sum_{y=0}^N \exp{\!\left( \mathrm{i} \vec{\gamma}\cdot\vec{y}  - \frac{\mathrm{i}}{2} \Gamma \right)} \ket{y}_C
\end{align} 
Now, applying the controlled-Z gate in Eq.\ref{eq:algorithm3}, we get:
\begin{align}
	\ket{y}_C \ket{\psi}_S 
    &\rightarrow \ket{y}_C R_z^{\otimes n}(y_1 \beta_1) \cdots R_z^{\otimes n}(y_k \beta_k) \ket{\psi}_S \notag \\
    &= \ket{y}_C R_z^{\otimes n} (y_1\beta_1 + \cdots + y_k \beta_k) \ket{\psi}_S \notag \\
    &= \ket{y}_C R_z^{\otimes n} \!\left(\sum_{j=1}^k y_j\beta_j \right) \ket{\psi}_S \notag \\
    &= \ket{y}_C R_z^{\otimes n} \!\left(\vec{y}\cdot\vec{\beta} \right) \ket{\psi}_S,
\end{align}
resulting in:
\begin{align}
	\ket{\Psi} = R_z^{\otimes n} \left(\vec{y}\cdot \vec{\beta}\right) \ket{\psi}_S \otimes \frac{1}{\sqrt{N+1}} \sum_{y=0}^N \exp{\!\left( \mathrm{i} \vec{\gamma}\cdot\vec{y}  - \frac{\mathrm{i}}{2} \Gamma \right)} \ket{y}_C
\end{align}
Here, 
\begin{align}
    \vec{\gamma}\cdot\vec{y} &= \sum_{j=1}^k \gamma_j y_j =\frac{n \pi}{N+1} \sum_{j=1}^k y_j 2^{j-1} = \frac{n \pi y}{N+1}, \\
    \vec{\beta}\cdot\vec{y} &= \sum_{j=1}^k \beta_j y_j =\frac{2 \pi}{N+1} \sum_{j=1}^k y_j 2^{j-1} = \frac{2 \pi y}{N+1}.
\end{align}
we get,
\begin{equation}
	\ket{\Psi} = \ket{y}_C \\ \otimes R_z^{\otimes n} \!\left( \frac{2 \pi y}{N+1}\right) \ket{\psi}_S \otimes \frac{1}{\sqrt{N+1}} \exp{\!\left(  - \frac{\mathrm{i}}{2} \Gamma \right)} \sum_{y=0}^N \exp{\!\left( \frac{\mathrm{i}n\pi y}{N+1} \right)} .
\end{equation}
Note that:
\begin{equation}
    R_z^{\otimes n}\!\left(\frac{2 \pi y}{N+1}\right) = \exp\! \left(\frac{-\mathrm{i}n \pi y}{N+1} \right) \sum_{x=0}^n \exp\! \left( \frac{2\pi \mathrm{i} xy}{N+1} \right) P_x.
\end{equation}
After the controlled-Z gate, the state becomes:
\begin{equation}
	\ket{\Psi} = \frac{1}{\sqrt{N+1}} \exp{\!\left(  - \frac{\mathrm{i}}{2} \Gamma \right)} \sum_{y=0}^N \sum_{x=0}^n \exp\! \left( \frac{2\pi \mathrm{i} xy}{N+1} \right) P_x\ket{\psi}_S\ket{y}_C .
\end{equation} 
Upon applying IQFT to the control register, we obtain:
\begin{equation}
	\frac{1}{N+1} \exp{\!\left(  - \frac{\mathrm{i}}{2} \Gamma \right)} \sum_{z,y,x=0}^N  \exp\! \left( \frac{2\pi \mathrm{i} (x-z) y}{N+1} \right) \\ \ket{z}_C P_x \ket{\psi}_S.
\end{equation}
Using the summation identity:
\begin{align}
	S = \sum_{y=0}^N  \exp\! \left( \frac{2\pi \mathrm{i} (x-z) y}{N+1} \right) = (N+1)\delta_{xz}
\end{align}
the state becomes:
\begin{equation}
	\exp{\!\left(  - \frac{\mathrm{i}}{2} \Gamma \right)} \sum_{x=0}^N \ket{x}_C \otimes P_x\ket{\psi}_S
\end{equation}
Therefore, the probability of measuring Hamming weight $a$ is:
\begin{equation}
	p_{a} = \left| P_{a} \ket{\psi} \right|^{2}
	\label{eq:hamming_weight_probability}
\end{equation}
and the post-measurement state is:
\begin{equation}
	\ket{\bar{\psi}}  = \frac{P_{a} \ket{\psi}}{\sqrt{p_{a}}}.
	\label{eq:post_measurement_state}
\end{equation}

\begin{minted}[bgcolor=bg,

def HammingWeight(vector, num_shot=100):
  # num of physical qubits
  n = np.log2(vector.shape[0]).astype(np.int64) 
  # num of register qubits
  k = np.ceil(np.log2(n+1)).astype(np.int64)
  N = 2**k - 1

  qc = qiskit.QuantumCircuit(n + k, k) 
  # Phase 1
  # Register Qubits
  state_computation = qiskit.QuantumCircuit(k)
  for i in range(k):
    state_computation.h(i)
  # State Preparation
  state_preparation = qiskit.QuantumCircuit(n)
  state_preparation.initialize(vector, 
                      state_preparation.qubits)

  qc = qc.compose(state_computation, 
                   qubits=[i for i in range(0,k,1)])
  qc = qc.compose(state_preparation, 
                   qubits=[i for i in range(k,k+n,1)])

  # Phase 2
  for i in range(0,k):
    theta = n*np.pi/(N+1)*2**i
    qc.rz(theta, i)

  # Phase 3
  for i in range(k-1,-1,-1):
    theta = 2*np.pi/(N+1) * 2**i
    for j in range(k+n-1,k-1,-1):
      qc.crz(theta, i, j)
   
  # Phase 4
  # IQFT
  for i in range(k-1,-1,-1): # 0 -> k-1
    for j in range(k-1, i, -1):
      theta = -2*np.pi / 2**(j-i+1)
      qc.crz(theta, j, i)
    qc.h(i)      

  for i in range(k):
    qc.measure(i,k-i-1)
  
  result = MeasurementResultHandler(
    backend.run(qc, shots=num_shot)
    .result().data()["counts"], k)
  
  return qc, result
\end{minted}

\section*{Exercise 4}
\subsubsection*{Explain how the function in part 2 above can be used to prepare a Dicke state.}

The function from part 2 can be used to prepare a Dicke state through post-measurement. From the Eq. \ref{eq:post_measurement_state}, 
we see that projecting the initial state onto the space of fixed Hamming weight $x$, results in a linear combination of computational basis states with Hamming weight $x$.
For example, in a three-qubit system with $x=2$, the basis is $\{\ket{011}, \ket{101}, \ket{110}\}$. 
A Dicke state for an $n$-qubit system and Hamming weight $k$ is given by:
\begin{equation}
  \ket{D_{n,k}} = \frac{1}{\sqrt{{}_n C_{k}}} \sum_{i=1}^{{}_n C_{k}} \ket{e_{i}}
\end{equation}
where $\ket{e_i}$ are the basis elements with Hamming weight $k$ and there are ${}_n C_{k}$ such basis states. 
Since the coefficients of the Dicke state are equal, the initial state before measuring the Hamming weight has the same coefficients in the computational basis.

This initial state is simply the $+$ state:
\begin{equation}
	\ket{\psi} = \ket{++ \cdots +} = \frac{1}{\sqrt{2^n}} \sum_{i=0}^{2^n -1} \ket{i}
\end{equation}
where $i$ represents the bit string. After measuring the Hamming weight $k$, the post-measurement state $\ket{\bar{\psi}}$ becomes Dicke state $\ket{D_{n,k}}$.

\section*{Exercise 5}
Suppose we want to prepare the Dicke state with $n = 100$ and $k = 50$.

\subsubsection*{a) What is the probability that the state will be measured?}
Preparing + state, $\ket{\psi} = \ket{+}^{\otimes n}$, the probability that we measure Hamming weight $k$ is given in the Eq. \ref{eq:hamming_weight_probability}.
Among the computational basis, $\{\ket{0}, \ket{1}, \cdots, \ket{2^n -1}\}$, the number of basis whose Hamming weight is ${}_n C_{k}$. Then,
\begin{equation}
	p_{n,k} = \left| P_{k} \ket{\psi} \right|^{2} = \left| \frac{{}_n C_{k}}{\sqrt{2^n}} \right|^{2} = \frac{({}_n C_k)^{2}}{2^n}.
\end{equation}

Therefore, in $n=100, k=50$, we have $p_{n,k} \simeq 0.07959$. 

\subsubsection*{b) What is the minimum number of measurements required?}

At least 13 shots are required for the expectation value of the number of measurements in the Dicke state $ket{D_{n=100,k=50}}$ to exceed 1. 
With this, the probability of measuring the Dicke state $\ket{D_{n=100,k=50}}$ is approximately 66\%.


\subsubsection*{c) How many gates are required?}

In Phase 1, we prepare the control register $C$, consisting of 7 qubits in initial state $\ket{0}_C$.
We also prepare $\ket{\psi}$  the system qubits. As discussed in Exercise 4, we initialize the system qubits in the $\ket{+}$ state, requiring 100 Hadamard gate in this phase. 
In Phase 2, we apply 7 Hadamard gates to the control register.
In Phase 3, we apply 7 Rz gates to the control register qubits.
In Phase 4, we perform the controlled unitary gate from Eq. \ref{eq:algorithm3}. This requires 100 Controlled RZ gate per register qubit, therefore, we need 700 controlled Rz gates.
In Phase 5, we perform  IQFT on controlled register qubits. 
This phase involves 7 hadamard gate and it applies controlled Rz gate that act on qubit in $j$ as control and qubit $i$ as target, 
where $i<j \leq k$ and $i$ runs from 0 to 7. Thus, we need 7 hadamard gate and 21 controlled Rz gate. 
Therefore, we need 114 Hadamard gates, 721 controlled Z gate, and 7 Rz gates.
